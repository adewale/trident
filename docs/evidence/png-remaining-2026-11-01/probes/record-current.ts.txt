import{readFileSync,writeFileSync,copyFileSync,mkdirSync}from'node:fs'
import{docsExpectationsHash}from'/workspace/trident-png-remaining-completion/unified/docs-png-drawings.ts'
import{docsPngStaticProjection}from'/workspace/trident-png-remaining-completion/unified/docs-png-static-contract.ts'
import{evidenceBytesDigest}from'/workspace/trident-png-remaining-completion/portable/evidence-contract.ts'
const dir='/workspace/trident-png-remaining-2026-11-01',base='/workspace/trident-png-full-2026-10-31/png-content-final-effective-style',saved=JSON.parse(readFileSync(base+'/result.json','utf8')),sources=JSON.parse(readFileSync('/workspace/trident-png-full-2026-10-31/docs/inventory.json','utf8')).cases,contracts=JSON.parse(readFileSync('/workspace/trident-png-remaining-completion/corpus/docs-reviewed.json','utf8')),notes=JSON.parse(readFileSync(dir+'/review-notes.json','utf8'));mkdirSync(dir+'/drawings',{recursive:true})
const receipts=notes.map((n:any)=>{const old=saved.cases.find((r:any)=>r.index===n.index&&r.project===n.project),source=sources[n.index],contract=contracts.find((r:any)=>r.id===old.id),png=readFileSync(base+'/'+old.witness),sha=evidenceBytesDigest(png),packet=n.index+'-'+(n.project==='mermaid-js/mermaid'?'m':'f')+'.png',projection=docsPngStaticProjection(contract);if(sha!==old.pngSha256||projection)throw Error('Unexpected image/projection');copyFileSync(base+'/'+old.witness,dir+'/'+packet);copyFileSync(base+'/'+old.witness,dir+'/drawings/'+sha+'.png');return{schema:'trident-offline-png-image-review-v2',index:n.index,project:n.project,source:source.source,reviewedSourceContract:contract,actualImage:{path:base+'/'+old.witness,sha256:sha,width:png.readUInt32BE(16),height:png.readUInt32BE(20)},inspection:{basis:'source-and-actual-image',method:'Fresh direct view_image inspection of actual retained PNG at native size against the complete current source contract; independently approved source/pixel measurements recorded where required',packetSha256:sha,packet,checkedAt:new Date().toISOString()},review:{id:old.id,sourceSha256:contract.sourceSha256,pngSha256:sha,expectationsSha256:docsExpectationsHash(contract),status:n.status,reviewedBy:'OpenAI Codex independent offline full-source and actual-image review, 2026-11-01',review:n.review}}})
writeFileSync(dir+'/reviews-current.jsonl',receipts.map((r:any)=>JSON.stringify(r)).join('\n')+'\n',{flag:'wx'});console.log(JSON.stringify({fresh:receipts.length}))
